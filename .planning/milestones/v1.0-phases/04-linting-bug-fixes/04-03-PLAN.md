- context: AudioContext instance (created unconditionally in init())
- buffers: Map of sound name to AudioBuffer
- unlocked: boolean (iOS audio unlock tracking)

2. Create `src/composables/useManagedTimeout.js`:
   ```javascript
   import { onUnmounted } from 'vue';

   export function useManagedTimeout() {
     const timeouts = new Set();

     function managedSetTimeout(fn, delay) {
       const id = setTimeout(() => {
         timeouts.delete(id);
         fn();
       }, delay);
       timeouts.add(id);
       return id;
     }

     function clearManagedTimeout(id) {
       clearTimeout(id);
       timeouts.delete(id);
     }

     function clearAll() {
       timeouts.forEach(clearTimeout);
       timeouts.clear();
     }

     onUnmounted(clearAll);

     return {
       managedSetTimeout,
       clearManagedTimeout,
       clearAll,
     };
   }
   ```

   Note: This composable intentionally does NOT include managed intervals. The only setInterval in the app is the game timer in the Pinia store (which is not a component and already manages its own interval via clearInterval). Components only use setTimeout.

**FIX-09: Audio system graceful degradation**

In `src/store/gameStore.js`, rewrite the `audioManager` object:

1. Add a `ready` property initialized to `false`.

2. Rewrite `init()` to wrap AudioContext creation in try-catch:
   ```javascript
   async init() {
     try {
       const AudioCtx = window.AudioContext || window.webkitAudioContext;
       if (!AudioCtx) {
         console.warn('AudioContext not available. Game will run silently.');
         return;
       }
       this.context = new AudioCtx();
       await Promise.allSettled([
         this.loadSound('stimulus', stimulusSoundUrl),
         this.loadSound('increment', incrementSoundUrl),
         this.loadSound('strike', strikeSoundUrl),
       ]);
       this.ready = true;
     } catch (e) {
       console.warn('Audio initialization failed. Game will run silently.', e);
       this.context = null;
       this.ready = false;
     }
   },
   ```

   Key changes:
   - Check for AudioContext existence before calling `new`
   - Use `Promise.allSettled` instead of `Promise.all` so partial sound loading succeeds
   - Set `this.ready = true` only after successful init
   - Catch any error and set ready to false

3. Rewrite `play()` to check `ready` state and wrap in try-catch:
   ```javascript
   play(soundName) {
     if (!this.ready || !this.context || !this.buffers[soundName]) return;
     try {
       if (this.context.state === 'suspended') {
         this.context.resume();
       }
       const source = this.context.createBufferSource();
       source.buffer = this.buffers[soundName];
       source.connect(this.context.destination);
       source.start(0);
     } catch (e) {
       console.warn(`Audio playback failed for "${soundName}":`, e);
     }
   }
   ```

4. Update `unlock()` to also check `ready`:
   ```javascript
   unlock() {
     if (this.unlocked || !this.ready || !this.context) return;
     if (this.context.state === 'suspended') {
       this.context.resume();
     }
     this.unlocked = true;
   },
   ```

**FIX-08: Global error handler**

```javascript
// Global error handlers
app.config.errorHandler = (err, instance, info) => {

};

window.onerror = (message, source, lineno, colno, error) => {
  console.error('[Global Error]', { message, source, lineno, colno, error });

  console.error('[Unhandled Promise Rejection]', event.reason);
});
```

Place these between `app.use(pinia);` and `app.mount('#app');`. This establishes the error handler before the app renders, catching any errors during mount and runtime.
